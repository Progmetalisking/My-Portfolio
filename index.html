# Edge Telemetry Anomaly Detection Filter
import numpy as np

def detect_sensor_drift(readings, threshold=2.8):
    mean, std = np.mean(readings), np.std(readings)
    z_scores = [(x - mean) / (std + 1e-5) for x in readings]
    return [idx for idx, z in enumerate(z_scores) if abs(z) > threshold]
